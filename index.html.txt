        function getTimeAgo(timestamp) {
            const seconds = Math.floor((Date.now() - timestamp) / 1000);
            if (seconds < 60) return 'Just now';
            const minutes = Math.floor(seconds / 60);
            if (minutes < 60) return `${minutes}m ago`;
            const hours = Math.floor(minutes / 60);
            if (hours < 24) return `${hours}h ago`;
            const days = Math.floor(hours / 24);
            return `${days}d ago`;
        }

        // Technical Analysis Calculations
        function calculateIndicators(prices) {
            const rsi = calculateRSI(prices, 14);
            const macd = calculateMACD(prices);
            const bb = calculateBollingerBands(prices);
            const ma50 = calculateMA(prices, 50);
            const ma200 = calculateMA(prices, 200);
            
            return { rsi, macd, bb, ma50, ma200 };
        }

        function calculateRSI(prices, period) {
            if (prices.length < period + 1) return 50;
            
            let gains = 0, losses = 0;
            for (let i = 1; i <= period; i++) {
                const diff = prices[prices.length - i] - prices[prices.length - i - 1];
                if (diff >= 0) gains += diff;
                else losses -= diff;
            }
            
            const avgGain = gains / period;
            const avgLoss = losses / period;
            const rs = avgGain / (avgLoss || 1);
            return 100 - (100 / (1 + rs));
        }

        function calculateMACD(prices) {
            const ema12 = calculateEMA(prices, 12);
            const ema26 = calculateEMA(prices, 26);
            const macdLine = ema12 - ema26;
            const signalLine = calculateEMA([...Array(9).fill(macdLine), macdLine], 9);
            return {
                value: macdLine,
                signal: signalLine,
                histogram: macdLine - signalLine
            };
        }

        function calculateEMA(prices, period) {
            const k = 2 / (period + 1);
            let ema = prices.slice(0, period).reduce((a, b) => a + b, 0) / period;
            for (let i = period; i < prices.length; i++) {
                ema = prices[i] * k + ema * (1 - k);
            }
            return ema;
        }

        function calculateBollingerBands(prices, period = 20) {
            const slice = prices.slice(-period);
            const sma = slice.reduce((a, b) => a + b, 0) / period;
            const squaredDiffs = slice.map(p => Math.pow(p - sma, 2));
            const stdDev = Math.sqrt(squaredDiffs.reduce((a, b) => a + b, 0) / period);
            
            return {
                upper: sma + (stdDev * 2),
                middle: sma,
                lower: sma - (stdDev * 2),
                width: (stdDev * 2) / sma,
                position: (prices[prices.length - 1] - (sma - stdDev * 2)) / ((stdDev * 4) || 1)
            };
        }

        function calculateMA(prices, period) {
            const slice = prices.slice(-period);
            return slice.reduce((a, b) => a + b, 0) / slice.length;
        }

        function calculateVolatility(prices) {
            const returns = [];
            for (let i = 1; i < prices.length; i++) {
                returns.push(Math.log(prices[i][1] / prices[i-1][1]));
            }
            const mean = returns.reduce((a, b) => a + b, 0) / returns.length;
            const variance = returns.reduce((a, b) => a + Math.pow(b - mean, 2), 0) / returns.length;
            return Math.sqrt(variance) * Math.sqrt(365) * 100;
        }

        function generateMonteCarloPredictions(prices, days) {
            const returns = [];
            for (let i = 1; i < prices.length; i++) {
                returns.push(Math.log(prices[i] / prices[i-1]));
            }
            
            const mean = returns.reduce((a, b) => a + b, 0) / returns.length;
            const variance = returns.reduce((a, b) => a + Math.pow(b - mean, 2), 0) / returns.length;
            const stdDev = Math.sqrt(variance);
            
            const simulations = 1000;
            const results = [];
            
            for (let sim = 0; sim < simulations; sim++) {
                let price = prices[prices.length - 1];
                const path = [];
                for (let day = 0; day < days; day++) {
                    const shock = randomNormal() * stdDev + mean;
                    price *= Math.exp(shock);
                    path.push(price);
                }
                results.push(path);
            }
            
            const meanPath = [];
            const stdDevPath = [];
            
            for (let day = 0; day < days; day++) {
                const dayPrices = results.map(r => r[day]);
                const dayMean = dayPrices.reduce((a, b) => a + b, 0) / simulations;
                const dayVariance = dayPrices.reduce((a, b) => a + Math.pow(b - dayMean, 2), 0) / simulations;
                meanPath.push(dayMean);
                stdDevPath.push(Math.sqrt(dayVariance));
            }
            
            return { mean: meanPath, stdDev: stdDevPath };
        }

        function randomNormal() {
            let u = 0, v = 0;
            while(u === 0) u = Math.random();
            while(v === 0) v = Math.random();
            return Math.sqrt(-2.0 * Math.log(u)) * Math.cos(2.0 * Math.PI * v);
        }

        function calculatePredictions(marketData) {
            const prices = marketData.prices.map(p => p[1]);
            const lastPrice = prices[prices.length - 1];
            const predictions = generateMonteCarloPredictions(prices, 90);
            
            const pred7d = predictions.mean[6];
            const pred30d = predictions.mean[29];
            const pred90d = predictions.mean[89];
            
            const confidence = calculatePredictionConfidence(prices, predictions);
            
            updatePredictionDisplay('pred7d', pred7d, lastPrice);
            updatePredictionDisplay('pred30d', pred30d, lastPrice);
            updatePredictionDisplay('pred90d', pred90d, lastPrice);
            
            document.getElementById('confidenceScore').textContent = confidence + '%';
            document.getElementById('confidenceBar').style.width = confidence + '%';
        }

        function updatePredictionDisplay(id, value, current) {
            const el = document.getElementById(id);
            const change = ((value / current - 1) * 100);
            el.textContent = formatPrice(value) + ` (${change >= 0 ? '+' : ''}${change.toFixed(1)}%)`;
            el.className = `text-xl font-bold font-mono ${change >= 0 ? 'text-emerald-400' : 'text-rose-400'}`;
        }

        function calculatePredictionConfidence(prices, predictions) {
            const volatility = calculateVolatility(prices.map((p, i) => [i, p]));
            const trendStrength = Math.abs(calculateRSI(prices, 14) - 50) / 50;
            const volumeConsistency = 0.7;
            
            const confidence = Math.min(95, Math.max(30, 
                100 - volatility * 2 + trendStrength * 20 + volumeConsistency * 10
            ));
            return Math.round(confidence);
        }

        function calculatePotential(coin, marketData) {
            const prices = marketData.prices.map(p => p[1]);
            const indicators = calculateIndicators(prices);
            
            const momentumScore = Math.min(100, Math.max(0, indicators.rsi));
            const trendScore = indicators.ma50 > indicators.ma200 ? 70 : 30;
            const volatilityScore = Math.min(100, calculateVolatility(marketData.prices) * 2);
            const marketCapScore = Math.min(100, 100 - Math.log10(coin.market_data.market_cap.usd || 1) * 10);
            const recoveryScore = Math.min(100, (coin.market_data.current_price.usd / (coin.market_data.ath.usd || 1)) * 100);
            
            const totalScore = Math.round(
                momentumScore * 0.25 + 
                trendScore * 0.25 + 
                (100 - volatilityScore) * 0.2 + 
                marketCapScore * 0.15 + 
                recoveryScore * 0.15
            );
            
            const circle = document.getElementById('potentialCircle');
            const circumference = 2 * Math.PI * 56;
            const offset = circumference - (totalScore / 100) * circumference;
            circle.style.strokeDashoffset = offset;
            
            document.getElementById('potentialScore').textContent = totalScore;
            
            const risk = totalScore > 70 ? 'High' : totalScore > 40 ? 'Medium' : 'Low';
            document.getElementById('riskLevel').textContent = risk;
            document.getElementById('riskLevel').className = `font-medium ${risk === 'High' ? 'text-rose-400' : risk === 'Medium' ? 'text-yellow-400' : 'text-emerald-400'}`;
            
            document.getElementById('volatilityLevel').textContent = volatilityScore > 80 ? 'Extreme' : volatilityScore > 50 ? 'High' : 'Moderate';
            document.getElementById('momentumLevel').textContent = indicators.rsi > 60 ? 'Strong' : indicators.rsi < 40 ? 'Weak' : 'Neutral';
            
            const lastPrice = prices[prices.length - 1];
            const atr = calculateATR(marketData.prices);
            
            document.getElementById('bullTarget').textContent = formatPrice(lastPrice * 1.3);
            document.getElementById('baseTarget').textContent = formatPrice(lastPrice * 1.05);
            document.getElementById('bearTarget').textContent = formatPrice(lastPrice * 0.85);
        }

        function calculateATR(prices) {
            if (prices.length < 14) return prices[prices.length-1][1] * 0.05;
            const trs = [];
            for (let i = 1; i < Math.min(14, prices.length); i++) {
                trs.push(Math.abs(prices[i][1] - prices[i-1][1]));
            }
            return trs.reduce((a, b) => a + b, 0) / trs.length;
        }

        function formatPrice(price) {
            if (price >= 1) return '$' + price.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
            if (price >= 0.01) return '$' + price.toFixed(4);
            return '$' + price.toFixed(6);
        }

        function formatLargeNumber(num) {
            if (num >= 1e12) return '$' + (num / 1e12).toFixed(2) + 'T';
            if (num >= 1e9) return '$' + (num / 1e9).toFixed(2) + 'B';
            if (num >= 1e6) return '$' + (num / 1e6).toFixed(2) + 'M';
            if (num >= 1e3) return '$' + (num / 1e3).toFixed(2) + 'K';
            return '$' + num.toFixed(2);
        }
